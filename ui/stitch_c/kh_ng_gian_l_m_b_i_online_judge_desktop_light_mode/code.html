Xét một thuật toán nhận vào số nguyên dương n. Nếu n là số chẵn, thuật toán sẽ chia n cho 2. Nếu n là số lẻ, thuật toán sẽ nhân n với 3 rồi cộng thêm 1.

Thuật toán lặp lại quy trình trên liên tục cho đến khi giá trị của n = 1. Ví dụ thực tế với khởi đầu n = 3, chuỗi dãy số sinh ra theo từng bước là:

// Complexity: O(k) steps, O(1) auxiliary space
using namespace std;
int main() {
// Tối ưu tốc độ luồng xuất nhập I/O
ios_base::sync_with_stdio(false);
cin.tie(nullptr);
// Bắt buộc dùng 64-bit int để tránh tràn số khi 3n + 1 vượt 2^31 - 1
long long n;
if (!(cin >> n)) return 0;
cout << n;
while (n != 1) {
if (n % 2 == 0) {
n /= 2;
} else {
n = 3 * n + 1;
}
cout << " " << n;
}
cout << "\n";
return 0;
}
